# 微信小程序无感行为追踪技术方案

## 一、技术方案概述

### 1.1 核心技术路线

| 技术方案 | 原理 | 适用场景 | 代表方案 |
|---------|------|---------|---------|
| **全埋点(无痕埋点)** | 代理App/Page/Component原生方法，自动采集生命周期事件 | 通用行为追踪 | 神策SDK、腾讯有数 |
| **代码埋点** | 手动在关键位置调用track接口 | 精细化业务事件 | 自研SDK |
| **可视化埋点** | 通过可视化工具配置采集点 | 无需发版的采集 | 神策可视化 |
| **混合埋点** | 全埋点+代码埋点结合 | 生产环境推荐 | 大多数企业方案 |

### 1.2 推荐技术栈

```
前端采集层：微信小程序原生SDK / Taro插件
传输层：wx.request / 批量上报
存储层：ClickHouse / MySQL + Redis
分析层：Flink实时计算 / 离线分析
可视化层：ECharts / AntV / Grafana
```

## 二、无感数据采集实现方案

### 2.1 核心原理：代理原生方法

通过重写微信小程序的 `App`、`Page`、`Component` 三个核心注册方法，拦截生命周期函数和用户交互事件，实现无感知数据采集。

```javascript
// 核心实现原理
const originalApp = App;
const originalPage = Page;

App = function(options) {
  // 代理App生命周期
  options = proxyAppOptions(options);
  return originalApp(options);
};

Page = function(options) {
  // 代理Page生命周期和自定义事件
  options = proxyPageOptions(options);
  return originalPage(options);
};
```

### 2.2 自动采集事件清单

| 事件类型 | 事件名称 | 采集时机 | 数据内容 |
|---------|---------|---------|---------|
| App生命周期 | $MPLaunch | 小程序启动 | 启动来源、场景值 |
| App生命周期 | $MPShow | 小程序显示 | 场景值、路径 |
| App生命周期 | $MPHide | 小程序隐藏 | 停留时长 |
| Page生命周期 | $MPViewScreen | 页面浏览 | 页面路径、标题 |
| Page生命周期 | $MPPageLeave | 页面离开 | 停留时长 |
| 用户交互 | $MPClick | 元素点击 | 元素信息、位置 |
| 用户交互 | $MPShare | 分享 | 分享渠道、路径 |
| 用户交互 | $MPPullDownRefresh | 下拉刷新 | 页面路径 |
| 用户交互 | $MPReachBottom | 上拉触底 | 页面路径 |

### 2.3 实现代码示例

```javascript
// tracker.js - 全埋点SDK核心
class MiniProgramTracker {
  constructor(config) {
    this.config = config;
    this.queue = [];
    this.cid = this.getOrCreateCID();
  }

  // 初始化，重写原生方法
  init() {
    this.proxyApp();
    this.proxyPage();
    this.proxyComponent();
    this.proxyRequest();
    this.proxyRoute();
  }

  // 代理App
  proxyApp() {
    const originalApp = App;
    const tracker = this;
    
    App = function(options) {
      // 代理onLaunch
      const originalLaunch = options.onLaunch;
      options.onLaunch = function(res) {
        tracker.track('app_launch', {
          path: getCurrentPages()[0]?.route,
          scene: res.scene,
          timestamp: Date.now()
        });
        originalLaunch?.call(this, res);
      };

      // 代理onShow
      const originalShow = options.onShow;
      options.onShow = function(res) {
        tracker.track('app_show', {
          scene: res.scene,
          path: res.path,
          timestamp: Date.now()
        });
        originalShow?.call(this, res);
      };

      // 代理onHide
      const originalHide = options.onHide;
      options.onHide = function() {
        tracker.track('app_hide', {
          timestamp: Date.now()
        });
        originalHide?.call(this);
      };

      return originalApp(options);
    };
  }

  // 代理Page
  proxyPage() {
    const originalPage = Page;
    const tracker = this;
    
    Page = function(options) {
      // 代理onLoad
      const originalLoad = options.onLoad;
      options.onLoad = function(query) {
        this._pageLoadTime = Date.now();
        tracker.track('page_load', {
          route: this.route,
          query: query,
          timestamp: Date.now()
        });
        originalLoad?.call(this, query);
      };

      // 代理onShow
      const originalShow = options.onShow;
      options.onShow = function() {
        this._pageShowTime = Date.now();
        tracker.track('page_show', {
          route: this.route,
          timestamp: Date.now()
        });
        originalShow?.call(this);
      };

      // 代理onHide
      const originalHide = options.onHide;
      options.onHide = function() {
        const duration = Date.now() - (this._pageShowTime || Date.now());
        tracker.track('page_hide', {
          route: this.route,
          duration: duration,
          timestamp: Date.now()
        });
        originalHide?.call(this);
      };

      // 代理onUnload
      const originalUnload = options.onUnload;
      options.onUnload = function() {
        const duration = Date.now() - (this._pageLoadTime || Date.now());
        tracker.track('page_unload', {
          route: this.route,
          duration: duration,
          timestamp: Date.now()
        });
        originalUnload?.call(this);
      };

      // 代理所有自定义事件（点击等）
      this.proxyPageEvents(options);

      return originalPage(options);
    };
  }

  // 代理页面自定义事件
  proxyPageEvents(options) {
    const tracker = this;
    const lifeCycleMethods = [
      'onLoad', 'onShow', 'onReady', 'onHide', 'onUnload',
      'onPullDownRefresh', 'onReachBottom', 'onShareAppMessage',
      'onShareTimeline', 'onPageScroll', 'onResize', 'onTabItemTap'
    ];

    for (let key in options) {
      if (typeof options[key] === 'function' && !lifeCycleMethods.includes(key)) {
        const originalFn = options[key];
        options[key] = function(e) {
          // 检测点击事件
          if (e && e.type === 'tap') {
            tracker.track('element_click', {
              route: this.route,
              element: e.currentTarget?.id || e.currentTarget?.dataset,
              timestamp: Date.now()
            });
          }
          return originalFn.call(this, e);
        };
      }
    }
  }

  // 数据上报
  track(event, data) {
    const payload = {
      event,
      data,
      cid: this.cid,
      uid: this.getUID(),
      timestamp: Date.now(),
      platform: 'wechat_miniprogram'
    };

    this.queue.push(payload);
    this.scheduleFlush();
  }

  // 批量上报
  scheduleFlush() {
    if (this.flushTimer) return;
    
    this.flushTimer = setTimeout(() => {
      this.flush();
      this.flushTimer = null;
    }, this.config.batchDelay || 3000);
  }

  flush() {
    if (this.queue.length === 0) return;

    const data = [...this.queue];
    this.queue = [];

    wx.request({
      url: this.config.serverUrl,
      method: 'POST',
      data: { events: data },
      success: () => {},
      fail: (err) => {
        // 失败时缓存到本地
        this.cacheEvents(data);
      }
    });
  }

  // 本地缓存
  cacheEvents(events) {
    try {
      const cached = wx.getStorageSync('tracker_cache') || [];
      cached.push(...events);
      wx.setStorageSync('tracker_cache', cached.slice(-1000)); // 最多缓存1000条
    } catch (e) {}
  }

  getOrCreateCID() {
    let cid = wx.getStorageSync('tracker_cid');
    if (!cid) {
      cid = this.generateUUID();
      wx.setStorageSync('tracker_cid', cid);
    }
    return cid;
  }

  getUID() {
    return wx.getStorageSync('tracker_uid') || 'anonymous';
  }

  generateUUID() {
    return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
      const r = Math.random() * 16 | 0;
      const v = c === 'x' ? r : (r & 0x3 | 0x8);
      return v.toString(16);
    });
  }
}

// 导出单例
module.exports = new MiniProgramTracker({
  serverUrl: 'https://your-server.com/api/track',
  batchDelay: 3000
});
```

## 三、行前-行中-行后三阶段行为记录

### 3.1 行前阶段（出发前）

| 采集点 | 采集内容 | 实现方式 |
|-------|---------|---------|
| 启动来源 | 场景值(scene)、来源App | App.onLaunch |
| 首次访问 | 新老用户标识、设备信息 | 本地存储判断 |
| 预加载行为 | 预加载页面、资源请求 | wx.preloadPage |
| 分享入口 | 分享来源、分享者ID | 页面参数解析 |

```javascript
// 行前阶段采集
App({
  onLaunch(options) {
    // 记录启动场景
    const preBehavior = {
      scene: options.scene,
      path: options.path,
      query: options.query,
      shareTicket: options.shareTicket,
      referrerInfo: options.referrerInfo,
      timestamp: Date.now(),
      isFirstVisit: !wx.getStorageSync('has_visited'),
      deviceInfo: this.getDeviceInfo()
    };
    
    tracker.track('pre_trip', preBehavior);
    
    if (preBehavior.isFirstVisit) {
      wx.setStorageSync('has_visited', true);
    }
  },
  
  getDeviceInfo() {
    try {
      const info = wx.getSystemInfoSync();
      return {
        brand: info.brand,
        model: info.model,
        system: info.system,
        platform: info.platform,
        SDKVersion: info.SDKVersion
      };
    } catch (e) {
      return {};
    }
  }
});
```

### 3.2 行中阶段（使用中）

| 采集点 | 采集内容 | 实现方式 |
|-------|---------|---------|
| 页面浏览 | 路径、停留时长、滚动深度 | Page生命周期 |
| 功能使用 | 功能模块、操作次数 | 代码埋点 |
| 交互行为 | 点击、滑动、输入 | 事件代理 |
| 业务操作 | 提交、支付、收藏 | 业务埋点 |
| 性能数据 | 加载时间、渲染时间 | 性能API |
| 错误信息 | JS错误、网络错误 | onError监听 |

```javascript
// 行中阶段采集
Page({
  onLoad() {
    this.startTime = Date.now();
    this.scrollDepth = 0;
    
    // 监听滚动深度
    this._onPageScroll = (e) => {
      this.scrollDepth = Math.max(this.scrollDepth, e.scrollTop);
    };
  },
  
  onShow() {
    this.showTime = Date.now();
    tracker.track('page_view', {
      route: this.route,
      query: this.options
    });
  },
  
  onHide() {
    const duration = Date.now() - this.showTime;
    tracker.track('page_leave', {
      route: this.route,
      duration: duration,
      scrollDepth: this.scrollDepth
    });
  },
  
  // 业务埋点示例
  handleOrderSubmit(e) {
    const orderData = e.detail.value;
    
    // 业务埋点
    tracker.track('order_submit', {
      orderId: orderData.orderId,
      amount: orderData.amount,
      items: orderData.items.length,
      route: this.route
    });
    
    // 业务逻辑...
  }
});
```

### 3.3 行后阶段（离开后）

| 采集点 | 采集内容 | 实现方式 |
|-------|---------|---------|
| 会话结束 | 总时长、页面访问数 | App.onHide |
| 分享行为 | 分享渠道、分享内容 | onShareAppMessage |
| 返回行为 | 返回来源、停留时长 | onUnload |
| 留存标记 | 最后访问时间、访问频次 | 本地存储 |

```javascript
// 行后阶段采集
App({
  onHide() {
    const sessionData = {
      sessionDuration: Date.now() - this.globalData.sessionStart,
      pagesVisited: getCurrentPages().length,
      lastPage: getCurrentPages().pop()?.route,
      timestamp: Date.now()
    };
    
    // 更新留存数据
    this.updateRetention();
    
    tracker.track('session_end', sessionData);
    
    // 强制刷新上报队列
    tracker.flush();
  },
  
  updateRetention() {
    const now = Date.now();
    const lastVisit = wx.getStorageSync('last_visit_time') || 0;
    const visitCount = wx.getStorageSync('visit_count') || 0;
    
    wx.setStorageSync('last_visit_time', now);
    wx.setStorageSync('visit_count', visitCount + 1);
    
    // 计算留存天数
    if (lastVisit > 0) {
      const daysSinceLastVisit = Math.floor((now - lastVisit) / (24 * 60 * 60 * 1000));
      wx.setStorageSync('days_since_last_visit', daysSinceLastVisit);
    }
  }
});
```

## 四、自动化数据采集实现

### 4.1 网络请求监控

```javascript
// 自动监控所有网络请求
function proxyRequest() {
  const originalRequest = wx.request;
  
  wx.request = function(options) {
    const startTime = Date.now();
    const requestId = generateRequestId();
    
    // 记录请求开始
    tracker.track('request_start', {
      requestId,
      url: options.url,
      method: options.method || 'GET',
      timestamp: startTime
    });
    
    const originalSuccess = options.success;
    const originalFail = options.fail;
    
    options.success = function(res) {
      tracker.track('request_success', {
        requestId,
        url: options.url,
        statusCode: res.statusCode,
        duration: Date.now() - startTime,
        dataSize: JSON.stringify(res.data).length
      });
      originalSuccess?.call(this, res);
    };
    
    options.fail = function(err) {
      tracker.track('request_fail', {
        requestId,
        url: options.url,
        error: err.errMsg,
        duration: Date.now() - startTime
      });
      originalFail?.call(this, err);
    };
    
    return originalRequest(options);
  };
}
```

### 4.2 路由跳转监控

```javascript
// 自动监控路由跳转
function proxyRoute() {
  const routeMethods = [
    'navigateTo', 'redirectTo', 'switchTab', 
    'reLaunch', 'navigateBack'
  ];
  
  routeMethods.forEach(method => {
    const original = wx[method];
    
    wx[method] = function(options) {
      const from = getCurrentPages().pop()?.route;
      
      tracker.track('route_change', {
        method,
        from,
        to: options.url,
        timestamp: Date.now()
      });
      
      return original.call(this, options);
    };
  });
}
```

### 4.3 错误自动捕获

```javascript
// 自动捕获全局错误
App({
  onError(error) {
    tracker.track('app_error', {
      message: error,
      timestamp: Date.now(),
      pages: getCurrentPages().map(p => p.route)
    });
  },
  
  onUnhandledRejection(res) {
    tracker.track('promise_rejection', {
      reason: res.reason?.message || res.reason,
      promise: res.promise,
      timestamp: Date.now()
    });
  }
});
```

## 六、开源方案推荐

### 6.1 神策数据SDK
- **GitHub**: https://github.com/sensorsdata/sa-sdk-miniprogram
- **特点**: 全埋点+代码埋点，功能完善

- **GitHub**: https://github.com/clklog/clklog

- **技术栈**: Java + Flink + ClickHouse + Vue

## 七、最佳实践建议

### 7.1 采集策略
1. **全埋点打底**: 使用全埋点采集基础行为数据
2. **代码埋点补充**: 关键业务节点使用代码埋点
3. **批量上报**: 减少网络请求，提升性能
4. **本地缓存**: 网络异常时缓存数据，恢复后补发

### 7.2 性能优化
1. **延迟初始化**: 在onLaunch后异步初始化SDK
2. **节流控制**: 高频事件(scroll等)做节流处理
3. **数据压缩**: 上报数据使用gzip压缩
4. **采样策略**: 高流量场景可设置采样率

### 7.3 隐私合规
1. **用户授权**: 首次使用需获取用户同意
2. **数据脱敏**: 敏感信息(手机号等)脱敏处理
3. **数据存储**: 遵循数据最小化原则
4. **GDPR合规**: 提供数据删除接口
